  try {
    const res = await window.storage.get("emolab-entries-v2", false);

function mode(arr) {
  if (!arr.length) return null;
  const counts = {};
  arr.forEach((v) => (counts[v] = (counts[v] || 0) + 1));
  return Object.entries(counts).sort((a, b) => b[1] - a[1])[0][0];
}

function ExercisePlayer({ exercise, accent, onFinish, onCancel }) {
  const [remaining, setRemaining] = useState(exercise.duration);
  const [stepIdx, setStepIdx] = useState(0);

  useEffect(() => {
    const t = setInterval(() => setRemaining((r) => (r <= 1 ? 0 : r - 1)), 1000);
    return () => clearInterval(t);
  }, []);

  useEffect(() => {
    const stepDur = exercise.duration / exercise.steps.length;
    const elapsed = exercise.duration - remaining;
    setStepIdx(Math.min(exercise.steps.length - 1, Math.floor(elapsed / stepDur)));
  }, [remaining, exercise]);

  const pct = 1 - remaining / exercise.duration;
  const shakeKeyframes = exercise.mode === "shake" && remaining > 0;

  return (
    <div style={{ display: "flex", flexDirection: "column", alignItems: "center", padding: "20px 20px 28px" }}>
      <style>{`@keyframes emo-shake { 0%,100%{transform:translateX(0)} 25%{transform:translateX(-4px) rotate(-2deg)} 75%{transform:translateX(4px) rotate(2deg)} }`}</style>
      <div style={{ alignSelf: "flex-start" }}><BackBtn onClick={onCancel} /></div>
      <div style={{ fontFamily: D, fontSize: 21, color: THEME.ink, textAlign: "center" }}>{exercise.title}</div>
      {exercise.intent && (
        <div style={{ fontFamily: F, fontSize: 12.5, color: THEME.ink, textAlign: "center", marginTop: 4, maxWidth: 280 }}>
          {exercise.intent}
        </div>
      )}

      <PoseDiagram exerciseId={exercise.id} accent={accent} phase={stepIdx} totalPhases={exercise.steps.length} />

      <div style={{ position: "relative", width: 140, height: 140, margin: "16px 0 16px" }}>
        <svg width="140" height="140" style={{ transform: "rotate(-90deg)" }}>
          <circle cx="70" cy="70" r="60" fill="none" stroke={`${accent}33`} strokeWidth="8" />
          <circle cx="70" cy="70" r="60" fill="none" stroke={accent} strokeWidth="8"
            strokeDasharray={2 * Math.PI * 60} strokeDashoffset={2 * Math.PI * 60 * (1 - pct)}
            strokeLinecap="round" style={{ transition: "stroke-dashoffset 1s linear" }} />
        </svg>
        <div style={{
          position: "absolute", inset: 0, display: "flex", alignItems: "center", justifyContent: "center",
          fontFamily: D, fontSize: 28, color: THEME.ink,
          animation: shakeKeyframes ? "emo-shake 0.3s infinite" : "none",
          transform: exercise.mode === "breathing" && !shakeKeyframes ? `scale(${1 + 0.12 * Math.sin(pct * Math.PI * 6)})` : "none",
          transition: "transform 1s ease",
        }}>{remaining}</div>
      </div>

      <div style={{ background: THEME.card, border: `1px solid ${THEME.line}`, borderRadius: 24, padding: "16px 20px", width: "100%", maxWidth: 300, textAlign: "center", fontFamily: F, fontSize: 14.5, color: THEME.ink, minHeight: 56 }}>
        {exercise.steps[stepIdx]}
      </div>

      {remaining === 0 ? (
        <button className="emo-btn" onClick={() => onFinish(true)} style={{ marginTop: 26, background: THEME.gradient, color: "#fff", border: "none", borderRadius: 999, padding: "12px 26px", fontFamily: FN, fontWeight: 700, fontSize: 15, cursor: "pointer", display: "flex", alignItems: "center", gap: 8, boxShadow: "0 4px 14px rgba(0,0,0,0.16)" }}>
          <Check size={18} /> Listo
        </button>
      ) : (
        <button className="emo-btn" onClick={() => onFinish(false)} style={{ marginTop: 26, background: "none", color: THEME.ink, border: `1px solid ${THEME.line}`, borderRadius: 999, padding: "10px 22px", fontFamily: F, fontSize: 13.5, cursor: "pointer" }}>
          Saltear
        </button>
      )}
    </div>
  );
}

function CheckIn({ onComplete }) {
  const [phase, setPhase] = useState("body");
  const [zones, setZones] = useState([]);
  const [sensations, setSensations] = useState([]);
  const [surface, setSurface] = useState(null);
  const [deepenText, setDeepenText] = useState("");
  const [granular, setGranular] = useState(null);
  const [necesidad, setNecesidad] = useState(null);
  const [chosenExercise, setChosenExercise] = useState(null);
  const [chosenState, setChosenState] = useState(null);

  const state = surface?.state || "sin_clasificar";
  const accent = STATES[state].color;

  const toggleZone = (id) => setZones((prev) => prev.includes(id) ? prev.filter((z) => z !== id) : [...prev, id]);
  const toggleSensation = (s) => setSensations((prev) => prev.includes(s) ? prev.filter((x) => x !== s) : [...prev, s]);

  const zoneLabels = zones.map((id) => ZONES.find((z) => z.id === id)?.label).filter(Boolean);
  const joinNatural = (arr) => arr.length <= 1 ? (arr[0] || "") : `${arr.slice(0, -1).join(", ")} y ${arr[arr.length - 1]}`;

  const persist = useCallback(async (interventionDone, exerciseTitle) => {
    const entry = {
      id: uid(), timestamp: Date.now(), zone: zones, sensation: sensations,
      surfaceEmotion: surface?.label || null, surfaceEmotionId: surface?.id || null, state,
      deepenText, granular, necesidad,
      interventionDone: !!interventionDone, exerciseTitle: exerciseTitle || null,
      flagged: false,
    };
    const current = await loadEntries();
    const updated = [entry, ...current];
    await saveEntries(updated);
    onComplete(updated);
  }, [zones, sensations, surface, deepenText, granular, necesidad, state, onComplete]);

  if (phase === "body") return (
    <div style={{ height: "100%", display: "flex", flexDirection: "column", boxSizing: "border-box", background: "#FFF4D5", overflow: "hidden" }}>
      <div style={{ padding: "22px 20px 4px", flexShrink: 0 }}>

      </div>
    </div>
  );

      <div style={{ marginTop: 22 }}>

            background: STATES[st].soft, fontFamily: F, fontSize: 15, color: THEME.ink, cursor: "pointer",
          }}>{STATES[st].label}</button>
        ))}
      </div>
    </div>
  );

  if (phase.startsWith("pickExercise2:")) {
    const st = phase.split(":")[1];
    return <ExercisePicker state={st} onCancel={() => setPhase("pickExerciseAny")} onPick={(ex) => { setChosenExercise(ex); setChosenState(st); setPhase("play"); }} />;
  }

  if (phase === "play") return (
    <ExercisePlayer exercise={chosenExercise} accent={STATES[chosenState]?.color || accent}
      onCancel={() => setPhase("decision")}
      onFinish={(completed) => persist(completed, chosenExercise.title)} />
  );

  return null;
}

/* sonido suave para marcar cambios de fase, sin archivos externos */
let sharedAudioCtx = null;
function playTone(freq = 420, duration = 150) {
  try {
    if (!sharedAudioCtx) sharedAudioCtx = new (window.AudioContext || window.webkitAudioContext)();
    const ctx = sharedAudioCtx;
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = "sine";
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(0.001, ctx.currentTime);
    gain.gain.linearRampToValueAtTime(0.06, ctx.currentTime + 0.02);
    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + duration / 1000);
    osc.connect(gain);
    gain.connect(ctx.destination);
    osc.start();
    osc.stop(ctx.currentTime + duration / 1000 + 0.02);
  } catch (e) { /* audio no disponible, seguimos sin sonido */ }
}
function hapticPulse() {
  try { if (navigator.vibrate) navigator.vibrate(25); } catch (e) { /* no disponible */ }
}

/* ---------- Sesión de respiración guiada ---------- */
function BreathingSession({ technique, totalSeconds, onFinish, onCancel }) {
  const [remaining, setRemaining] = useState(totalSeconds);
  const [phaseIdx, setPhaseIdx] = useState(0);
  const [phaseElapsed, setPhaseElapsed] = useState(0);

  useEffect(() => {
    hapticPulse();
    playTone(technique.phases[0].mode === "in" ? 520 : 380);
  }, []); // eslint-disable-line

  useEffect(() => {
    const t = setInterval(() => {
      setRemaining((r) => Math.max(0, r - 0.1));
      setPhaseElapsed((pe) => pe + 0.1);
    }, 100);
    return () => clearInterval(t);
  }, []);

  useEffect(() => {
    const phase = technique.phases[phaseIdx];
    if (phaseElapsed >= phase.seconds) {
      const nextIdx = (phaseIdx + 1) % technique.phases.length;
      setPhaseIdx(nextIdx);
      setPhaseElapsed(0);
      hapticPulse();
      playTone(technique.phases[nextIdx].mode === "in" ? 520 : technique.phases[nextIdx].mode === "out" ? 380 : 450);
    }
  }, [phaseElapsed, phaseIdx, technique]);

  const phase = technique.phases[phaseIdx];
  const phasePct = Math.min(1, phaseElapsed / phase.seconds);

  // calcula la escala del círculo: crece en 'in', se mantiene en 'hold', decrece en 'out'
  let scale;
  if (phase.mode === "in") scale = 0.62 + phasePct * 0.5;
  else if (phase.mode === "out") scale = 1.12 - phasePct * 0.5;
  else {
    // hold: mantiene la escala alcanzada al final de la fase anterior
    const prevIdx = (phaseIdx - 1 + technique.phases.length) % technique.phases.length;
    scale = technique.phases[prevIdx].mode === "in" ? 1.12 : 0.62;
  }

  const finished = remaining <= 0;

  useEffect(() => {
    if (finished) onFinish();
  }, [finished]); // eslint-disable-line

  const mm = Math.floor(remaining / 60);
  const ss = Math.floor(remaining % 60);

  return (
    <div style={{ display: "flex", flexDirection: "column", alignItems: "center", padding: "20px 20px 28px", height: "100%" }}>
      <div style={{ alignSelf: "flex-start" }}><BackBtn onClick={onCancel} /></div>
      <div style={{ fontFamily: D, fontSize: 19, color: THEME.ink, textAlign: "center" }}>{technique.title}</div>
      <div style={{ fontFamily: F, fontSize: 12.5, color: THEME.ink, marginTop: 2 }}>
        {String(mm).padStart(2, "0")}:{String(ss).padStart(2, "0")} restantes
      </div>

      <div style={{ position: "relative", width: 220, height: 220, margin: "30px 0", display: "flex", alignItems: "center", justifyContent: "center" }}>
        <div style={{
          width: 150, height: 150, borderRadius: "50%",
          background: `radial-gradient(circle, ${THEME.regulado}55 0%, ${THEME.regulado}25 60%, ${THEME.regulado}00 100%)`,
          transform: `scale(${scale})`, transition: "transform 0.1s linear",
          display: "flex", alignItems: "center", justifyContent: "center",
        }}>
          <div style={{
            width: 92, height: 92, borderRadius: "50%", background: THEME.regulado,
            display: "flex", alignItems: "center", justifyContent: "center",
          }}>
            <span style={{ fontFamily: D, fontSize: 16, color: "#fff" }}>{phase.label}</span>
          </div>
        </div>
      </div>

      <p style={{ fontFamily: F, fontSize: 12.5, color: THEME.ink, textAlign: "center", maxWidth: 260 }}>{technique.intent}</p>

      <button className="emo-btn" onClick={onCancel} style={{
        marginTop: 20, background: "none", border: `1px solid ${THEME.line}`, borderRadius: 999,
        padding: "10px 22px", fontFamily: F, fontSize: 13.5, color: THEME.ink, cursor: "pointer",
      }}>
        Terminar antes
      </button>
    </div>
  );
}

function SixSecondFlow({ onExit }) {
  const [step, setStep] = useState("intro"); // intro | pickTask | timer | closing
  const [task, setTask] = useState(null);

  if (step === "intro") return <SixSecondIntro onCancel={onExit} onStart={() => setStep("pickTask")} />;
  if (step === "pickTask") return <SixSecondTaskPicker onCancel={() => setStep("intro")} onPick={(t) => { setTask(t); setStep("timer"); }} />;
  if (step === "timer") return <SixSecondTimer task={task} onFinish={() => setStep("closing")} />;
  if (step === "closing") return (
    <SixSecondClosing onDone={async (helped) => {
      const session = { id: uid(), timestamp: Date.now(), task: task.id, taskLabel: task.label, helped };
      const current = await loadSixSecondSessions();
      await saveSixSecondSessions([session, ...current]);
      onExit();
    }} />
  );
  return null;
}

/* ---------- Autoevaluación de Inteligencia Emocional ---------- */
function IETest({ onDone, onCancel }) {
  const [answers, setAnswers] = useState({});
  const allAnswered = IE_ITEMS.every((it) => answers[it.id]);

  const submit = async () => {
    const branchScores = {};
    Object.keys(IE_BRANCHES).forEach((b) => {
      const items = IE_ITEMS.filter((it) => it.branch === b);
      const sum = items.reduce((acc, it) => acc + (answers[it.id] || 0), 0);
      branchScores[b] = sum / items.length;
    });
    const result = { id: uid(), timestamp: Date.now(), scores: branchScores };
    const current = await loadIeResults();
    await saveIeResults([result, ...current]);
    onDone(result);
  };

  return (
    <div style={{ padding: "24px 20px" }}>
      <BackBtn onClick={onCancel} />
      <h2 style={{ fontFamily: D, fontSize: 19, color: THEME.ink, marginBottom: 4 }}>Autoevaluación de IE</h2>
      <p style={{ fontFamily: F, fontSize: 12.5, color: THEME.ink, marginBottom: 18, lineHeight: 1.5 }}>
        Inspirada en el modelo de las 4 ramas de Mayer y Salovey. No reemplaza un test clínico validado (como el MSCEIT): es una herramienta reflexiva para conversar en terapia.
      </p>
      <div style={{ display: "flex", flexDirection: "column", gap: 18 }}>
        {IE_ITEMS.map((it) => (
          <div key={it.id}>
            <div style={{ fontFamily: F, fontSize: 14, color: THEME.ink, marginBottom: 8, lineHeight: 1.4 }}>{it.text}</div>
            <div style={{ display: "flex", gap: 6 }}>
              {[1, 2, 3, 4, 5].map((n) => (
                <button key={n} onClick={() => setAnswers((a) => ({ ...a, [it.id]: n }))} style={{
                  flex: 1, padding: "8px 0", borderRadius: 10, border: `2px solid ${answers[it.id] === n ? THEME.gold : THEME.line}`,
                  background: answers[it.id] === n ? `${THEME.gold}22` : THEME.card, fontFamily: F, fontSize: 13, color: THEME.ink, cursor: "pointer",
                }}>{n}</button>
              ))}
            </div>
            <div style={{ display: "flex", justifyContent: "space-between", marginTop: 3 }}>
              <span style={{ fontFamily: F, fontSize: 10, color: THEME.ink }}>Nunca</span>
              <span style={{ fontFamily: F, fontSize: 10, color: THEME.ink }}>Siempre</span>
            </div>
          </div>
        ))}
      </div>
      <div style={{ marginTop: 24 }}>
        <PrimaryButton disabled={!allAnswered} color={THEME.gold} onClick={submit}>Ver resultado</PrimaryButton>
      </div>
    </div>
  );
}

function ieCategory(avg) {
  if (avg < 2.5) return { label: "Terreno para explorar", desc: "Recién estás empezando a poner nombre y forma a esto. Es un buen momento para trabajarlo en terapia." };
  if (avg < 3.4) return { label: "En desarrollo", desc: "Ya reconocés algunas señales, pero todavía cuesta sostenerlas. Hay margen para afinar la percepción y la regulación." };
  if (avg < 4.3) return { label: "Con buen manejo", desc: "Manejás bastante bien tu mundo emocional, con algunas ramas más entrenadas que otras." };
  return { label: "Muy entrenada", desc: "Tenés un manejo fino de tus emociones en la mayoría de las ramas evaluadas." };
}

/* elige la emoción que más se repitió en la semana, a partir de los check-ins (misma lógica que la Nube) */
function dominantPrimaryThisWeek(entries) {
  const weekAgo = Date.now() - 7 * 24 * 3600 * 1000;
  const counts = {};
  entries.filter((e) => e.timestamp >= weekAgo).forEach((e) => {
    const primary = SURFACE_TO_PRIMARY[e.surfaceEmotionId];
    if (primary) counts[primary] = (counts[primary] || 0) + 1;
  });
  const sorted = Object.entries(counts).sort((a, b) => b[1] - a[1]);
  return sorted.length ? sorted[0][0] : null;
}

function Minijuego({ entries, onExit, onGoToRescate }) {
  const [phase, setPhase] = useState("intro"); // intro | level1..5 | result
  const [correctCount, setCorrectCount] = useState(0);
  const [finalPoints, setFinalPoints] = useState(0);
  const [rank, setRank] = useState(GAME_RANKS[0]);
  const [dominada, setDominada] = useState(false);
  const primary = useMemo(() => dominantPrimaryThisWeek(entries) || "miedo", [entries]);

  const nextLevel = { level1: "level2", level2: "level3", level3: "level4", level4: "level5" };

  const advance = (fromLevel, ok) => {
    const updatedCount = ok ? correctCount + 1 : correctCount;
    setCorrectCount(updatedCount);
    if (fromLevel === "level5") {
      finishGame(updatedCount);
    } else {
      setPhase(nextLevel[fromLevel]);
    }
  };

  const finishGame = async (finalCorrect) => {
    const progress = await loadGameProgress();
    const prev = progress[primary] || { points: 0, timesPlayed: 0, successCount: 0, dominada: false };
    const earned = finalCorrect * 20;
    const newSuccess = finalCorrect >= 3 ? prev.successCount + 1 : prev.successCount;
    const newProgress = {
      points: prev.points + earned,
      timesPlayed: prev.timesPlayed + 1,
      successCount: newSuccess,
      dominada: prev.dominada || newSuccess >= 3,
    };
    progress[primary] = newProgress;
    await saveGameProgress(progress);

    const totalPoints = Object.values(progress).reduce((s, p) => s + (p.points || 0), 0);
    setFinalPoints(earned);
    setRank(rankForPoints(totalPoints));
    setDominada(newProgress.dominada);
    setPhase("result");
  };

  if (phase === "intro") return <GameIntro primary={primary} onExit={onExit} onStart={() => setPhase("level1")} />;
  if (phase === "level1") return <GameLevel1 primary={primary} onResult={(ok) => advance("level1", ok)} />;
  if (phase === "level2") return <GameLevel2 primary={primary} onResult={(ok) => advance("level2", ok)} />;
  if (phase === "level3") return <GameLevel3 primary={primary} onResult={(ok) => advance("level3", ok)} onSuggestRescate={onGoToRescate} />;
  if (phase === "level4") return <GameLevel4 primary={primary} onResult={(ok) => advance("level4", ok)} />;
  if (phase === "level5") return <GameLevel5 primary={primary} onResult={(ok) => advance("level5", ok)} />;
  if (phase === "result") return <GameResult primary={primary} correctCount={correctCount} points={finalPoints} rank={rank} dominada={dominada} onFinish={onExit} />;
  return null;
}

function MiSesion({ entries, onToggleFlag, profile, onLogout, onGoToRescate }) {

    loadIeResults().then((r) => { if (r.length) setLastIe(r[0]); });

export default function EmolabApp() {
  const [tab, setTab] = useState("checkin");
  const [entries, setEntries] = useState([]);
  const [profile, setProfile] = useState(null);
  const [checkinKey, setCheckinKey] = useState(0);
  const [justSaved, setJustSaved] = useState(false);
  const [loading, setLoading] = useState(true);
  const [showWelcome, setShowWelcome] = useState(false);
  const [proScreen, setProScreen] = useState(null); // null | "login" | "panel"

  const loadProfile = () => window.storage.get("emolab-profile", false).then((r) => r ? JSON.parse(r.value) : null).catch(() => null);

  useEffect(() => {
    Promise.all([loadEntries(), window.storage.get("emolab-welcome-seen", false).catch(() => null), loadProfile()])
      .then(([e, seen, prof]) => {
        setEntries(e);
        setShowWelcome(!seen);
        setProfile(prof);
        setLoading(false);
      });
  }, []);

  const handleStart = async () => {
    setShowWelcome(false);
    try { await window.storage.set("emolab-welcome-seen", "true", false); } catch (e) { /* noop */ }
    loadProfile().then(setProfile);
  };

  const handleComplete = (updated) => {
    setEntries(updated);
    setJustSaved(true);
    setTimeout(() => setJustSaved(false), 2200);
    setTab("checkin");
    setCheckinKey((k) => k + 1);
  };

  const handleToggleFlag = async (id) => {
    const updated = entries.map((e) => (e.id === id ? { ...e, flagged: !e.flagged } : e));
    setEntries(updated);
    await saveEntries(updated);
  };

  const handleLogout = () => {
    setTab("checkin");
    setShowWelcome(true);
  };

  if (!loading && proScreen === "login") {
    return (
      <div className="emolab-frame" style={{ background: THEME.bg, height: 600, width: "100%", maxWidth: 420, margin: "0 auto", borderRadius: 24, overflow: "hidden", position: "relative", display: "flex", flexDirection: "column", boxShadow: "0 8px 40px rgba(0,0,0,0.12)" }}>
        <style>{`@import url('https://fonts.googleapis.com/css2?family=Varela+Round&display=swap');
    * { -webkit-tap-highlight-color: transparent; box-sizing: border-box; }
    .emolab-scroll { scrollbar-width: none; -ms-overflow-style: none; }
    .emolab-scroll::-webkit-scrollbar { display: none; width: 0; height: 0; }
    @media (max-width: 480px) {
      .emolab-frame { height: 100dvh !important; max-width: 100% !important; border-radius: 0 !important; box-shadow: none !important; margin: 0 !important; }
    }
    .emo-btn { transition: transform 0.15s ease, filter 0.15s ease, box-shadow 0.15s ease; }
    .emo-btn:hover { filter: brightness(1.04); }
    .emo-btn:active { transform: scale(0.97); filter: brightness(0.95); }
  `}</style>
        <ProLogin onCancel={() => setProScreen(null)} onSuccess={() => setProScreen("panel")} />
      </div>
    );
  }

  if (!loading && proScreen === "panel") {
    return (
      <div className="emolab-frame" style={{ background: THEME.bg, height: 600, width: "100%", maxWidth: 420, margin: "0 auto", borderRadius: 24, overflow: "hidden", position: "relative", display: "flex", flexDirection: "column", boxShadow: "0 8px 40px rgba(0,0,0,0.12)" }}>
        <style>{`@import url('https://fonts.googleapis.com/css2?family=Varela+Round&display=swap');
    * { -webkit-tap-highlight-color: transparent; box-sizing: border-box; }
    .emolab-scroll { scrollbar-width: none; -ms-overflow-style: none; }
    .emolab-scroll::-webkit-scrollbar { display: none; width: 0; height: 0; }
    @media (max-width: 480px) {
      .emolab-frame { height: 100dvh !important; max-width: 100% !important; border-radius: 0 !important; box-shadow: none !important; margin: 0 !important; }
    }
    .emo-btn { transition: transform 0.15s ease, filter 0.15s ease, box-shadow 0.15s ease; }
    .emo-btn:hover { filter: brightness(1.04); }
    .emo-btn:active { transform: scale(0.97); filter: brightness(0.95); }
  `}</style>
        <ProPanel onLogout={() => setProScreen(null)} />
      </div>
    );
  }

  if (!loading && showWelcome) {
    return (
      <div className="emolab-frame" style={{ background: THEME.bg, height: 600, width: "100%", maxWidth: 420, margin: "0 auto", borderRadius: 24, overflow: "hidden", position: "relative", display: "flex", flexDirection: "column", boxShadow: "0 8px 40px rgba(0,0,0,0.12)" }}>
        <style>{`@import url('https://fonts.googleapis.com/css2?family=Varela+Round&display=swap');
    * { -webkit-tap-highlight-color: transparent; box-sizing: border-box; }
    .emolab-scroll { scrollbar-width: none; -ms-overflow-style: none; }
    .emolab-scroll::-webkit-scrollbar { display: none; width: 0; height: 0; }
    @media (max-width: 480px) {
      .emolab-frame { height: 100dvh !important; max-width: 100% !important; border-radius: 0 !important; box-shadow: none !important; margin: 0 !important; }
    }
    .emo-btn { transition: transform 0.15s ease, filter 0.15s ease, box-shadow 0.15s ease; }
    .emo-btn:hover { filter: brightness(1.04); }
    .emo-btn:active { transform: scale(0.97); filter: brightness(0.95); }
  `}</style>
        <Welcome onStart={handleStart} onProLogin={() => setProScreen("login")} />
      </div>
    );
  }

  return (
    <div className="emolab-frame" style={{ background: THEME.bg, height: 600, width: "100%", maxWidth: 420, margin: "0 auto", borderRadius: 24, overflow: "hidden", position: "relative", display: "flex", flexDirection: "column", boxShadow: "0 8px 40px rgba(0,0,0,0.12)" }}>
      <style>{`@import url('https://fonts.googleapis.com/css2?family=Varela+Round&display=swap');
    * { -webkit-tap-highlight-color: transparent; box-sizing: border-box; }
    .emolab-scroll { scrollbar-width: none; -ms-overflow-style: none; }
    .emolab-scroll::-webkit-scrollbar { display: none; width: 0; height: 0; }
    @media (max-width: 480px) {
      .emolab-frame { height: 100dvh !important; max-width: 100% !important; border-radius: 0 !important; box-shadow: none !important; margin: 0 !important; }
    }
    .emo-btn { transition: transform 0.15s ease, filter 0.15s ease, box-shadow 0.15s ease; }
    .emo-btn:hover { filter: brightness(1.04); }
    .emo-btn:active { transform: scale(0.97); filter: brightness(0.95); }
  `}</style>

      <div style={{ padding: "16px 20px 10px", display: "flex", alignItems: "center", flexShrink: 0, position: "relative" }}>
        <div style={{ width: 40, height: 40, borderRadius: "50%", background: THEME.gradient, display: "flex", alignItems: "center", justifyContent: "center", padding: 3, boxSizing: "border-box" }}>
          <div style={{ width: "100%", height: "100%", borderRadius: "50%", background: "#fff", display: "flex", alignItems: "center", justifyContent: "center" }}>
            <img src={LOGO_ICON} alt="emolab" style={{ width: 24, height: 24, objectFit: "contain" }} />
          </div>
        </div>
        <div style={{ position: "absolute", left: 0, right: 0, bottom: 0, height: 3, background: THEME.gradient, opacity: 0.5 }} />
      </div>

      {justSaved && (
        <div style={{ position: "absolute", top: 60, left: "50%", transform: "translateX(-50%)", zIndex: 10, background: THEME.ink, color: "#fff", padding: "8px 16px", borderRadius: 999, fontFamily: F, fontSize: 13, display: "flex", alignItems: "center", gap: 6 }}>
          <Check size={14} /> Check-in guardado
        </div>
      )}

      <div className="emolab-scroll" style={{ flex: 1, overflowY: "auto" }}>
        {loading ? (
          <div style={{ padding: 40, textAlign: "center", fontFamily: F, color: THEME.ink }}>Cargando...</div>
        ) : tab === "checkin" ? <CheckIn key={checkinKey} onComplete={handleComplete} />
        : tab === "rescate" ? <Rescate />
        : tab === "sesion" ? <MiSesion entries={entries} onToggleFlag={handleToggleFlag} profile={profile} onLogout={handleLogout} onGoToRescate={() => setTab("rescate")} />
        : <Glosario entries={entries} onGoToRescate={() => setTab("rescate")} />}
      </div>

      <div style={{ display: "flex", borderTop: `1px solid ${THEME.line}`, flexShrink: 0 }}>
        {TABS.map((t) => {
          const Icon = t.icon;
          const isActive = tab === t.id;
          return (
            <button key={t.id} onClick={() => setTab(t.id)} style={{
              flex: 1, padding: "12px 0 10px", background: "none", border: "none", cursor: "pointer",
              display: "flex", flexDirection: "column", alignItems: "center", gap: 3, position: "relative",
            }}>
              <div style={{ position: "absolute", top: 0, left: "50%", transform: "translateX(-50%)", width: 28, height: 3, borderRadius: 999, background: THEME.gradient, opacity: isActive ? 1 : 0 }} />
              <Icon size={17} color={isActive ? THEME.ink : THEME.inkSoft} />
              <span style={{ fontFamily: F, fontSize: 10.5, fontWeight: isActive ? 600 : 400, color: isActive ? THEME.ink : THEME.inkSoft }}>{t.label}</span>
            </button>
          );
        })}
      </div>
    </div>
  );
}
